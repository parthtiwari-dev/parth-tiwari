Unfold the exact final SQL

Committed Day 6 output · no SQL is executed here.

WITH customer_order_counts AS (
    SELECT customer_id, COUNT(order_id) AS order_count
    FROM orders
    GROUP BY customer_id
),
average_order_count AS (
    SELECT AVG(order_count) AS avg_count
    FROM customer_order_counts
)
SELECT c.customer_id, c.name, c.email, c.created_at, c.country, c.lifetime_value
FROM customers c
JOIN customer_order_counts coc ON c.customer_id = coc.customer_id
JOIN (SELECT AVG(order_count) AS avg_count FROM customer_order_counts) aoc ON coc.order_count > aoc.avg_count
LIMIT 1000;
